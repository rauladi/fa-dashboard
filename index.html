async function boot(){
  loadLS();
  initPreloadedData();
  let liveData = null;
  if(location.protocol !== 'file:'){
    try{
      const r = await fetch('./data.json?_='+Date.now());
      if(r.ok){
        liveData = await r.json();
        console.log('Loaded data.json — generated:', liveData.generated);
      }
    }catch(e){ console.log('data.json not available, using built-in data'); }
  }
  if(liveData && liveData.stocks){
    ANNUALISATION = liveData.annualisation || {};
    const ts = Date.now();
    S.stocks = {...BASE_STOCKS};
    S.custom = {};
    S.deleted = {};
    Object.entries(liveData.stocks).forEach(([sym, sd])=>{
      const hasData = Array.isArray(sd.revenue) && sd.revenue.some(v=>v!==null);
      S.stocks[sym] = { name: sd.name||sym, exchange: sd.exchange||'NYSE', currency: sd.currency||'B USD' };
      S.data[sym] = {
        totalAsset: sd.totalAsset, cash: sd.cash, totalDebt: sd.totalDebt, totalEquity: sd.totalEquity,
        revenue: sd.revenue, grossProfit: sd.grossProfit, netProfit: sd.netProfit, eps: sd.eps, dps: sd.dps,
        fetchedAt: ts, source: sd.source==='yfinance'?'yfinance':'built-in', ticker: sd.ticker, generated: liveData.generated
      };
      S.status[sym] = hasData ? 'ok' : 'idle';
    });
    const genDate = new Date(liveData.generated).toLocaleDateString();
    const n = Object.keys(liveData.stocks).length;
    const hSub = document.getElementById('hSub');
    if(hSub) hSub.textContent = `ASX · IDX · NYSE/NASDAQ — Auto-refreshed ${genDate} · ${n} stocks · ${YEARS[0]}–${YEARS[N_YEARS-1]}`;
    if(!S.stocks[S.sel]) S.sel = Object.keys(S.stocks)[0] || 'BHP';
  } else {
    const hSub = document.getElementById('hSub');
    if(hSub) hSub.textContent = `ASX · IDX · NYSE/NASDAQ — Built-in data · ${YEARS[0]}–${YEARS[N_YEARS-1]} · Host on GitHub Pages for auto-refresh`;
    if(!S.stocks[S.sel]) S.sel = 'BHP';
  }
  Object.keys(S.stocks).forEach(k=>{ if(!S.status[k]) S.status[k]=S.data[k]?'ok':'idle'; });
  render();
}
